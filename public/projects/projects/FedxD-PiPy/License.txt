import os
import shutil

def remove_venv_folders(base_dir):
    # Walk through the directory tree
    for root, dirs, files in os.walk(base_dir, topdown=False):
        if 'venv' in dirs:
            venv_path = os.path.join(root, 'venv')
            print(f"Removing {venv_path}...")
            try:
                shutil.rmtree(venv_path)
                print(f"Successfully removed: {venv_path}")
            except Exception as e:
                print(f"Error while removing {venv_path}: {e}")

if __name__ == "__main__":
    # Replace this with the root directory where your projects are stored
    base_directory = 'D:\New folder\Projects\Python'  # Change this to your base folder
    remove_venv_folders(base_directory)
